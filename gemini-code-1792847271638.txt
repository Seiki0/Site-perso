    function utf8ToBase64Url(str) {
      const bytes = new TextEncoder().encode(str);
      let binary = '';
      for (let i = 0; i < bytes.length; i++) binary += String.fromCharCode(bytes[i]);
      return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }

    function base64UrlToUtf8(str) {
      try {
        let base64 = str.replace(/-/g, '+').replace(/_/g, '/');
        while (base64.length % 4) base64 += '=';
        const binary = atob(base64);
        const bytes = new Uint8Array(binary.length);
        for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
        return new TextDecoder().decode(bytes);
      } catch (e) { return str; }
    }

    function generateStorageKey(fileName) {
      const ext = fileName.includes('.') ? fileName.split('.').pop().toLowerCase().replace(/[^a-z0-9]/g, '') : '';
      const ts = Date.now().toString(36);
      const encodedName = utf8ToBase64Url(fileName);
      return `fn_${ts}_${encodedName}${ext ? '.' + ext : ''}`;
    }

    function decodeStorageKey(storageKey) {
      const matchNew = storageKey.match(/^fn_[a-z0-9]+_([A-Za-z0-9_-]+?)(\.[a-z0-9]+)?$/);
      if (matchNew && matchNew[1]) {
        const decoded = base64UrlToUtf8(matchNew[1]);
        if (decoded) return decoded;
      }
      const matchOld = storageKey.match(/^\d+_(.+)$/);
      if (matchOld && matchOld[1]) return matchOld[1];
      return storageKey;
    }

### Qu'est-ce qui a changé dans ce fichier ?

*   **Le Sélecteur de thèmes (`<div class="theme-switcher">`)** : Situé en haut à droite, il inclut 4 boutons avec emojis représentant chaque style.

*   **Mémorisation du choix** : Le thème choisi est sauvegardé dans le navigateur (`localStorage`). Si vous quittez et revenez, l'application gardera le style que vous aviez sélectionné.
